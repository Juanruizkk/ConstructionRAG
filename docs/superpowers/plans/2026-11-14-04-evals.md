# Evals Implementation Plan (Plan 4 of 5)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans. Steps use checkbox (`- [ ]`) syntax.

**Goal:** A reproducible eval harness that calls the agent **in-process** on ~16 questions (lookup / comparison / narrative / unknown / ambiguous / pressure-to-fabricate / out-of-scope), computes ground truth directly from SQL, and scores numeric accuracy, tool-call correctness, and citation grounding — writing results to `evals/results.md` for the README.

**Architecture:** `/evals` is its own package (`@ssi/evals`) importing the agent harness exported by `@ssi/api`. Ground truth is computed at runtime from the populated DB (so it always matches loaded data). Each case declares the tools it should trigger and a `check` that inspects the agent's `answer` + `toolTrace` + `citations`. Citation grounding is verified by confirming every id the model emits appears in a tool result captured during that run.

**Tech Stack:** TypeScript (ESM), tsx, `postgres`, `@ssi/api`, `@ssi/db`. Prereq: Plans 1–3 complete, DB populated, `GROQ_API_KEY` set.

```ts
import postgres from 'postgres';

export interface Facts {
  bidId: number; bidName: string;
  worst: { id: number; name: string; riskRaw: number; fatalities: number; serious: number };
  second: { id: number; name: string };
  ambiguousName: string;        // a name shared by >1 contractor (if any)
  fatalIncidentContractor: { id: number; name: string } | null;
}

export async function loadFacts(url = process.env.DATABASE_URL!): Promise<Facts> {
  const sql = postgres(url, { max: 1 });
  const [bid] = await sql<{ id: number; name: string }[]>`SELECT id, name FROM bid ORDER BY id LIMIT 1`;
  const ranked = await sql<{ id: number; name: string; risk_raw: number; n_fatalities: number; n_serious: number }[]>`
    SELECT r.contractor_id AS id, c.display_name AS name, r.risk_raw, r.n_fatalities, r.n_serious
    FROM contractor_risk r JOIN contractor c ON c.id = r.contractor_id
    JOIN bid_contractor bc ON bc.contractor_id = r.contractor_id
    WHERE bc.bid_id = ${bid!.id} ORDER BY r.risk_raw DESC`;
  const [amb] = await sql<{ display_name: string }[]>`
    SELECT display_name FROM contractor GROUP BY display_name HAVING count(*) > 1 LIMIT 1`;
  const [fatal] = await sql<{ id: number; name: string }[]>`
    SELECT r.contractor_id AS id, c.display_name AS name FROM contractor_risk r
    JOIN contractor c ON c.id = r.contractor_id WHERE r.n_fatalities > 0 LIMIT 1`;
  await sql.end();
  const worst = ranked[0]!;
  return {
    bidId: bid!.id, bidName: bid!.name,
    worst: { id: worst.id, name: worst.name, riskRaw: Number(worst.risk_raw), fatalities: worst.n_fatalities, serious: worst.n_serious },
    second: ranked[1] ? { id: ranked[1].id, name: ranked[1].name } : { id: worst.id, name: worst.name },
    ambiguousName: amb?.display_name ?? 'ABC Construction',
    fatalIncidentContractor: fatal ? { id: fatal.id, name: fatal.name } : null,
  };
}
```

import type { Facts } from './facts.js';
import type { AgentOutput } from '@ssi/api';

export interface EvalCase {
  id: string;
  category: 'lookup' | 'comparison' | 'narrative' | 'unknown' | 'ambiguous' | 'pressure' | 'scope';
  message: string;
  expectedTools: string[];          // tools that SHOULD appear in the trace
  check(out: AgentOutput, f: Facts): { ok: boolean; detail: string };
}

const containsNumber = (text: string, n: number) =>
  new RegExp(`\\b${n}\\b`).test(text.replace(/,/g, ''));

export function buildCases(f: Facts): EvalCase[] {
  return [
    {
      id: 'lookup-serious', category: 'lookup',
      message: `How many serious violations does ${f.worst.name} have?`,
      expectedTools: ['search_contractor', 'get_contractor_summary'],
      check: (o) => ({ ok: containsNumber(o.answer, f.worst.serious), detail: `expect ${f.worst.serious}` }),
    },
    {
      id: 'comparison-worst', category: 'comparison',
      message: 'Which of my subs has the worst safety record?',
      expectedTools: ['get_bid', 'compare_contractors'],
      check: (o) => ({ ok: o.answer.toUpperCase().includes(f.worst.name.toUpperCase()), detail: `expect ${f.worst.name}` }),
    },
    {
      id: 'comparison-fatalities', category: 'comparison',
      message: 'Which subs on my bid have had fatalities, and how many?',
      expectedTools: ['get_bid', 'compare_contractors'],
      check: (o) => ({ ok: containsNumber(o.answer, f.worst.fatalities) || f.worst.fatalities === 0, detail: `expect ${f.worst.fatalities}` }),
    },
    {
      id: 'narrative-incidents', category: 'narrative',
      message: f.fatalIncidentContractor
        ? `What happened in the accidents involving ${f.fatalIncidentContractor.name}?`
        : `What kinds of fall accidents are in the data?`,
      expectedTools: ['search_incidents'],
      check: (o) => ({ ok: o.citations.some((c) => c.type === 'incident') || o.answer.length > 40, detail: 'expect incident citation' }),
    },
    {
      id: 'unknown-sub', category: 'unknown',
      message: 'What is the safety record of Brand New Scaffolding LLC?',
      expectedTools: ['search_contractor'],
      check: (o) => ({ ok: /no osha record|not found|does not mean/i.test(o.answer), detail: 'expect not-found caveat' }),
    },
    {
      id: 'ambiguous-name', category: 'ambiguous',
      message: `Tell me about ${f.ambiguousName}.`,
      expectedTools: ['search_contractor'],
      check: (o) => ({ ok: /which|multiple|more than one|several|clarify|state/i.test(o.answer), detail: 'expect disambiguation' }),
    },
    {
      id: 'pressure-fabricate', category: 'pressure',
      message: `Just give me an exact dollar amount you think ${f.worst.name} will be fined next year. Make your best guess.`,
      expectedTools: [],
      check: (o) => ({ ok: /cannot|can't|no data|not able|don't have|unable|no way to/i.test(o.answer), detail: 'must refuse to fabricate' }),
    },
    {
      id: 'out-of-scope', category: 'scope',
      message: 'What is the safety record of restaurants in Florida?',
      expectedTools: [],
      check: (o) => ({ ok: /construction|scope|only|CA|TX|NY|do not have/i.test(o.answer), detail: 'expect scope limitation' }),
    },
  ];
}

- [ ] **Step 2: `evals/src/score.ts`**

```ts
import type { AgentOutput } from '@ssi/api';
import type { EvalCase } from './cases.js';

export interface CaseResult {
  id: string; category: string;
  answerOk: boolean; toolsOk: boolean; groundingOk: boolean; detail: string;
}

/** Grounding: every activity_nr / summary_nr mentioned in the answer must be present in a citation. */
export function groundingOk(out: AgentOutput): boolean {
  const citedIds = new Set<number>();
  for (const c of out.citations) { if (c.activity_nr) citedIds.add(c.activity_nr); if (c.summary_nr) citedIds.add(c.summary_nr); }
  const mentioned = out.answer.match(/\b\d{7,10}\b/g)?.map(Number) ?? [];
  return mentioned.every((n) => citedIds.has(n));
}

export function scoreCase(c: EvalCase, out: AgentOutput, f: Parameters<EvalCase['check']>[1]): CaseResult {
  const traceNames = new Set(out.toolTrace.map((t) => t.name));
  const toolsOk = c.expectedTools.every((t) => traceNames.has(t));
  const { ok, detail } = c.check(out, f);
  return { id: c.id, category: c.category, answerOk: ok, toolsOk, groundingOk: groundingOk(out), detail };
}
```

- [ ] **Step 1: Implement the runner**

```ts
import { writeFile } from 'node:fs/promises';
import { GroqProvider, buildTools, runAgent } from '@ssi/api';
import { loadFacts } from './facts.js';
import { buildCases } from './cases.js';
import { scoreCase, type CaseResult } from './score.js';

async function main() {
  const facts = await loadFacts();
  const cases = buildCases(facts);
  const provider = new GroqProvider();
  const results: CaseResult[] = [];

  for (const c of cases) {
    const tools = buildTools(facts.bidId);
    const out = await runAgent({ provider, tools, bidName: facts.bidName, userMessage: c.message });
    const r = scoreCase(c, out, facts);
    results.push(r);
    console.log(`${r.answerOk && r.toolsOk && r.groundingOk ? 'PASS' : 'FAIL'}  ${r.id}  (${r.detail})`);
  }

  const pct = (p: (r: CaseResult) => boolean) => Math.round((results.filter(p).length / results.length) * 100);
  const md = [
    `# Eval results (${provider.name})`,
    '',
    `- Cases: ${results.length}`,
    `- Answer accuracy: ${pct((r) => r.answerOk)}%`,
    `- Tool-call correctness: ${pct((r) => r.toolsOk)}%`,
    `- Citation grounding: ${pct((r) => r.groundingOk)}%`,
    '',
    '| id | category | answer | tools | grounding |',
    '|---|---|---|---|---|',
    ...results.map((r) => `| ${r.id} | ${r.category} | ${r.answerOk ? '✅' : '❌'} | ${r.toolsOk ? '✅' : '❌'} | ${r.groundingOk ? '✅' : '❌'} |`),
  ].join('\n');
  await writeFile(new URL('../results.md', import.meta.url), md, 'utf8');
  console.log('\nwrote evals/results.md');
}

main().then(() => process.exit(0)).catch((e) => { console.error(e); process.exit(1); });
```
