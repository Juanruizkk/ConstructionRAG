# ETL Implementation Plan (Plan 2 of 5)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans. Steps use checkbox (`- [ ]`) syntax.

**Goal:** Populate the database from the DOL API: scoped fetch (construction / CA·TX·NY / ~10y) with a resumable cache, load inspections→violations→accidents, run entity resolution, build & embed accident narratives, and refresh the `contractor_risk` materialized view.

**Architecture:** ETL lives in `apps/api/src/etl` as CLI commands (`pnpm etl:*`) that import repositories from `@ssi/db`. A deterministic raw cache (`data/raw/<endpoint>/<hash>.json`) makes fetches idempotent and resumable. Dependent tables are fetched from the in-scope `activity_nr` set via the `in` operator in batches, each batch paginated with `offset`. Entity resolution normalizes names, blocks by state, matches on trigram + distinctive-token rules, clusters with Union-Find + cohesion validation, then assigns `inspection.contractor_id`. Embeddings use a local ONNX model behind an `EmbeddingProvider` interface.

**Tech Stack:** TypeScript (ESM), `@huggingface/transformers` (`Xenova/bge-small-en-v1.5`, 384-dim), `@ssi/db`, `postgres`, zod (config), vitest, tsx. Prereq: Plan 1 complete, Docker DB up, `DATABASE_URL` + `API_KEY_ODP` set.

**Fixed ETL order (do not reorder):** fetch → clean/upsert → entity resolution → assign `inspection.contractor_id` → recompute `accident_chunk.contractor_ids` → embed (only new/changed text) → `REFRESH contractor_risk`.

```ts
import { createHash } from 'node:crypto';
import { mkdir, readFile, writeFile } from 'node:fs/promises';
import { dirname, join } from 'node:path';

const RAW_DIR = join(process.cwd(), '..', '..', 'data', 'raw');

export function cacheKey(endpoint: string, filterObject: unknown, offset: number): string {
  const payload = JSON.stringify({ endpoint, filterObject, offset });
  const hash = createHash('sha256').update(payload).digest('hex').slice(0, 16);
  return join(RAW_DIR, endpoint, `${hash}.json`);
}

export async function readCache<T>(path: string): Promise<T | null> {
  try {
    return JSON.parse(await readFile(path, 'utf8')) as T;
  } catch {
    return null;
  }
}

export async function writeCache(path: string, data: unknown): Promise<void> {
  await mkdir(dirname(path), { recursive: true });
  await writeFile(path, JSON.stringify(data), 'utf8');
}
```

- [ ] **Step 1: Implement the client**

```ts
import { API_BASE, env } from '../config.js';
import { cacheKey, readCache, writeCache } from './cache.js';

type FilterObject = unknown;

async function sleep(ms: number) {
  return new Promise((r) => setTimeout(r, ms));
}

/** One page: cached by (endpoint, filter, offset). Retries on 429/5xx with backoff. */
async function fetchPage(
  endpoint: string,
  filterObject: FilterObject | undefined,
  offset: number,
  limit: number,
): Promise<Record<string, unknown>[]> {
  const path = cacheKey(endpoint, filterObject ?? null, offset);
  const cached = await readCache<Record<string, unknown>[]>(path);
  if (cached) return cached;

  const url = new URL(`${API_BASE}/get/osha/${endpoint}/json`);
  url.searchParams.set('X-API-KEY', env.API_KEY_ODP);
  url.searchParams.set('limit', String(limit));
  url.searchParams.set('offset', String(offset));
  if (filterObject) url.searchParams.set('filter_object', JSON.stringify(filterObject));

  let attempt = 0;
  for (;;) {
    const res = await fetch(url);
    if (res.ok) {
      const body = (await res.json()) as { data?: Record<string, unknown>[] };
      const rows = body.data ?? [];
      await writeCache(path, rows);
      return rows;
    }
    if ((res.status === 429 || res.status >= 500) && attempt < 5) {
      await sleep(2 ** attempt * 500);
      attempt += 1;
      continue;
    }
    throw new Error(`DOL ${endpoint} offset=${offset} -> ${res.status} ${await res.text()}`);
  }
}

/** Fetch ALL rows for a filter, paginating by offset until a short page. */
export async function fetchAll(
  endpoint: string,
  filterObject: FilterObject | undefined,
  limit: number,
): Promise<Record<string, unknown>[]> {
  const out: Record<string, unknown>[] = [];
  let offset = 0;
  for (;;) {
    const page = await fetchPage(endpoint, filterObject, offset, limit);
    out.push(...page);
    if (page.length < limit) break;
    offset += limit;
  }
  return out;
}

export function chunk<T>(arr: T[], size: number): T[][] {
  const res: T[][] = [];
  for (let i = 0; i < arr.length; i += size) res.push(arr.slice(i, i + size));
  return res;
}
```

```ts
export function reassemble(rows: Record<string, unknown>[]): Map<number, string> {
  const byId = new Map<number, { line: number; text: string }[]>();
  for (const r of rows) {
    const id = Number(r['summary_nr']);
    const line = Number(r['line_nr']) || 0;
    const text = String(r['abstract_text'] ?? '');
    const arr = byId.get(id) ?? [];
    arr.push({ line, text });
    byId.set(id, arr);
  }
  const out = new Map<number, string>();
  for (const [id, parts] of byId) {
    parts.sort((a, b) => a.line - b.line);
    out.set(id, parts.map((p) => p.text).join('').trim());
  }
  return out;
}
```

```ts
import { createClient, InspectionRepository, AccidentRepository } from '@ssi/db';
import { SCOPE } from '../../config.js';
import { fetchAll, chunk } from '../dol-client.js';
import { reassemble } from '../narratives.js';

export async function loadAccidents(): Promise<void> {
  const { db, queryClient } = createClient();
  const inspections = new InspectionRepository(db);
  const repo = new AccidentRepository(db);
  const inScope = new Set(await inspections.allActivityNrs());

  // 1) injuries whose employer inspection is in scope
  const injuries: Record<string, unknown>[] = [];
  for (const batch of chunk([...inScope], SCOPE.batchSize)) {
    const filter = { field: 'rel_insp_nr', operator: 'in', value: batch };
    injuries.push(...await fetchAll('accident_injury', filter, SCOPE.pageLimit));
  }
  let orphans = 0;
  const kept = injuries.filter((i) => {
    const ok = inScope.has(Number(i['rel_insp_nr']));
    if (!ok) orphans += 1;
    return ok;
  });
  const summaryNrs = [...new Set(kept.map((i) => Number(i['summary_nr'])))];

  // 2) accident + accident_abstract for those summary_nrs
  const accRows: Record<string, unknown>[] = [];
  const absRows: Record<string, unknown>[] = [];
  for (const batch of chunk(summaryNrs, SCOPE.batchSize)) {
    const filter = { field: 'summary_nr', operator: 'in', value: batch };
    accRows.push(...await fetchAll('accident', filter, SCOPE.pageLimit));
    absRows.push(...await fetchAll('accident_abstract', filter, SCOPE.pageLimit));
  }
  const narratives = reassemble(absRows);

  await repo.upsertAccidents(accRows.map((r) => ({
    summaryNr: Number(r['summary_nr']),
    eventDate: typeof r['event_date'] === 'string' ? (r['event_date'] as string).slice(0, 10) : null,
    eventDesc: (r['event_desc'] as string) ?? narratives.get(Number(r['summary_nr'])) ?? null,
    projectType: (r['project_type'] as string) ?? null,
    buildStories: Number(r['build_stories']) || null,
    nonbuildHt: Number(r['nonbuild_ht']) || null,
    fatalityFlag: String(r['fatality'] ?? '') === 'X',
  })));
  await repo.insertInjuries(kept.map((r) => ({
    summaryNr: Number(r['summary_nr']),
```

```ts
import { ER } from '../../config.js';

function levenshtein(a: string, b: string): number {
  const dp = Array.from({ length: a.length + 1 }, (_, i) => [i, ...Array(b.length).fill(0)]);
  for (let j = 0; j <= b.length; j++) dp[0]![j] = j;
  for (let i = 1; i <= a.length; i++)
    for (let j = 1; j <= b.length; j++)
      dp[i]![j] = Math.min(
        dp[i - 1]![j]! + 1, dp[i]![j - 1]! + 1,
        dp[i - 1]![j - 1]! + (a[i - 1] === b[j - 1] ? 0 : 1),
      );
  return dp[a.length]![b.length]!;
}

export function distinctiveTokens(norm: string): string[] {
  return norm.split(/\s+/).filter((t) => t && !ER.genericTokens.has(t));
}

export function distinctiveTokensMatch(aNorm: string, bNorm: string): boolean {
  const a = distinctiveTokens(aNorm);
  const b = distinctiveTokens(bNorm);
  for (const ta of a)
    for (const tb of b) {
      if (ta === tb) return true;
      if (ta.length >= 4 && tb.length >= 4 && levenshtein(ta, tb) <= 1) return true;
    }
  return false;
}
```

Uses a trigram-like Dice coefficient in-memory (mirrors pg_trgm closely enough for clustering; the DB trgm index is used at query time in Plan 3). Merge only when similarity ≥ floor AND distinctive tokens match; then validate each member against the canonical.
```ts
import { ER } from '../../config.js';
import { distinctiveTokensMatch } from './tokens.js';

function trigrams(s: string): Set<string> {
  const p = `  ${s} `;
  const g = new Set<string>();
  for (let i = 0; i < p.length - 2; i++) g.add(p.slice(i, i + 3));
  return g;
}
export function similarity(a: string, b: string): number {
  const A = trigrams(a), B = trigrams(b);
  let inter = 0;
  for (const g of A) if (B.has(g)) inter += 1;
  return (2 * inter) / (A.size + B.size || 1);
}

class DSU {
  parent: number[];
  constructor(n: number) { this.parent = Array.from({ length: n }, (_, i) => i); }
  find(x: number): number { return this.parent[x] === x ? x : (this.parent[x] = this.find(this.parent[x]!)); }
  union(a: number, b: number) { this.parent[this.find(a)] = this.find(b); }
}

export function clusterNames(norms: string[]): string[][] {
  const n = norms.length;
  const dsu = new DSU(n);
  for (let i = 0; i < n; i++)
    for (let j = i + 1; j < n; j++) {
      if (similarity(norms[i]!, norms[j]!) >= ER.trgmFloor && distinctiveTokensMatch(norms[i]!, norms[j]!)) {
        dsu.union(i, j);
      }
    }
  // group
  const groups = new Map<number, number[]>();
  for (let i = 0; i < n; i++) {
    const r = dsu.find(i);
    (groups.get(r) ?? groups.set(r, []).get(r)!).push(i);
  }
  // cohesion validation vs canonical (most frequent == first here; freq handled by caller)
  const result: string[][] = [];
  for (const idxs of groups.values()) {
    const canonical = norms[idxs[0]!]!;
    const kept: string[] = [];
    const rejected: string[] = [];
    for (const i of idxs) {
      const m = norms[i]!;
      if (m === canonical || similarity(m, canonical) >= ER.cohesionTrgm || distinctiveTokensMatch(m, canonical)) kept.push(m);
      else rejected.push(m);
    }
    result.push(kept);
    for (const r of rejected) result.push([r]); // split off
  }
  return result;
}
```

```ts
export async function resolveEntities(): Promise<void> {
  const { db, queryClient } = createClient();
  const repo = new ContractorRepository(db);

  // pull all inspections (raw name + state + activity_nr)
  const rows = await queryClient<{ activity_nr: number; estab_name_raw: string | null; site_state: string | null }[]>`
    SELECT activity_nr, estab_name_raw, site_state FROM inspection`;

  // group by state (blocking)
  const byState = new Map<string, { nr: number; raw: string; norm: string }[]>();
  for (const r of rows) {
    const norm = normalizeName(r.estab_name_raw ?? '');
    if (!norm) continue;
    const st = r.site_state ?? 'NA';
    (byState.get(st) ?? byState.set(st, []).get(st)!).push({ nr: r.activity_nr, raw: r.estab_name_raw ?? '', norm });
  }

  let big = 0;
  for (const [state, recs] of byState) {
    const uniqueNorms = [...new Set(recs.map((r) => r.norm))];
    const clusters = clusterNames(uniqueNorms);
    for (const cluster of clusters) {
      if (cluster.length > 50) big += 1; // suspicious mega-cluster (README)
      const members = recs.filter((r) => cluster.includes(r.norm));
      // canonical = most frequent norm; display = most frequent raw
      const freq = new Map<string, number>();
      for (const m of members) freq.set(m.norm, (freq.get(m.norm) ?? 0) + 1);
      const canonical = [...freq].sort((a, b) => b[1] - a[1])[0]![0];
      const rawFreq = new Map<string, number>();
      for (const m of members) rawFreq.set(m.raw, (rawFreq.get(m.raw) ?? 0) + 1);
      const display = [...rawFreq].sort((a, b) => b[1] - a[1])[0]![0];

      const contractorId = await repo.createCanonical({
        canonicalName: canonical, displayName: display, normName: canonical, primaryState: state,
      });
      const aliasFreq = new Map<string, { raw: string; count: number }>();
      for (const m of members) {
        const e = aliasFreq.get(m.norm) ?? { raw: m.raw, count: 0 };
        e.count += 1; aliasFreq.set(m.norm, e);
      }
      for (const [norm, info] of aliasFreq) {
        await repo.addAlias({
          contractorId, rawName: info.raw, normName: norm, occurrences: info.count,
          matchMethod: norm === canonical ? 'exact' : 'trgm', confidence: '0.9000',
        });
      }
      for (const m of members) await repo.setInspectionContractor(m.nr, contractorId);
    }
  }
```

```sql
  SELECT DISTINCT summary_nr, rel_insp_nr FROM accident_injury WHERE rel_insp_nr IS NOT NULL;

DROP MATERIALIZED VIEW IF EXISTS contractor_risk;
CREATE MATERIALIZED VIEW contractor_risk AS
WITH v AS (
  SELECT i.contractor_id,
         vi.viol_type,
         vi.final_order_date,
         POWER(0.5, GREATEST(0, (CURRENT_DATE - vi.issuance_date)) / 365.0 / 5.0) AS decay
  FROM violation vi JOIN inspection i ON i.activity_nr = vi.activity_nr
  WHERE i.contractor_id IS NOT NULL
),
vagg AS (
  SELECT contractor_id,
    COUNT(*) FILTER (WHERE viol_type='S') AS n_serious,
    COUNT(*) FILTER (WHERE viol_type='W') AS n_willful,
    COUNT(*) FILTER (WHERE viol_type='R') AS n_repeat,
    COUNT(*) FILTER (WHERE viol_type='O') AS n_other,
    COUNT(*) FILTER (WHERE viol_type='U') AS n_unclassified,
    COUNT(*) FILTER (WHERE final_order_date IS NULL) AS n_pending_contested,
    SUM(decay * CASE viol_type WHEN 'W' THEN 10 WHEN 'R' THEN 7 WHEN 'S' THEN 3 ELSE 1 END) AS vss
  FROM v GROUP BY contractor_id
),
-- does the injury's inspection carry any S/W/R citation?
cited AS (
  SELECT ai.summary_nr, ai.rel_insp_nr,
         bool_or(vi.viol_type IN ('S','W','R')) AS has_swr
  FROM accident_injury ai
  LEFT JOIN violation vi ON vi.activity_nr = ai.rel_insp_nr
  GROUP BY ai.summary_nr, ai.rel_insp_nr
),
h AS (
  SELECT i.contractor_id, ai.degree_of_inj,
         POWER(0.5, GREATEST(0,(CURRENT_DATE - a.event_date))/365.0/5.0) AS decay,
         COALESCE(c.has_swr, false) AS has_swr
  FROM accident_injury ai
  JOIN inspection i ON i.activity_nr = ai.rel_insp_nr AND i.contractor_id IS NOT NULL
  JOIN accident a ON a.summary_nr = ai.summary_nr
  LEFT JOIN cited c ON c.summary_nr = ai.summary_nr AND c.rel_insp_nr = ai.rel_insp_nr
),
hagg AS (
  SELECT contractor_id,
    COUNT(*) FILTER (WHERE degree_of_inj='1') AS n_fatalities,
    COUNT(*) FILTER (WHERE degree_of_inj='1' AND has_swr) AS n_fatalities_cited,
    COUNT(*) FILTER (WHERE degree_of_inj='1' AND NOT has_swr) AS n_fatalities_no_citation,
    COUNT(*) FILTER (WHERE degree_of_inj='2') AS n_hospitalizations,
    SUM(COALESCE(decay,0)
        * CASE degree_of_inj WHEN '1' THEN 50 WHEN '2' THEN 15 WHEN '3' THEN 3 ELSE 0 END
        * CASE WHEN has_swr THEN 1.0 ELSE 0.3 END) AS hs
  FROM h GROUP BY contractor_id
),

  COALESCE(insp.n_inspections_clean,0) AS n_inspections_clean,
  COALESCE(vagg.n_serious,0) AS n_serious,

  COALESCE(vagg.n_unclassified,0) AS n_unclassified,
  COALESCE(vagg.n_pending_contested,0) AS n_pending_contested,
  COALESCE((SELECT SUM(current_penalty::numeric) FROM violation vi JOIN inspection i2 ON i2.activity_nr=vi.activity_nr WHERE i2.contractor_id=c.id),0) AS total_current_penalty,
  COALESCE(hagg.n_fatalities,0) AS n_fatalities,
  COALESCE(hagg.n_fatalities_cited,0) AS n_fatalities_cited,
  COALESCE(hagg.n_fatalities_no_citation,0) AS n_fatalities_no_citation,
  COALESCE(hagg.n_hospitalizations,0) AS n_hospitalizations,
  insp.last_inspection_date,
  (10 * COALESCE(vagg.vss,0) / GREATEST(COALESCE(insp.n_inspections,0),3) + COALESCE(hagg.hs,0)) AS risk_raw,
  LEAST(100, ROUND(10 * COALESCE(vagg.vss,0) / GREATEST(COALESCE(insp.n_inspections,0),3) + COALESCE(hagg.hs,0)))::int AS risk_score,
  NOW() AS score_as_of
FROM contractor c
LEFT JOIN insp ON insp.contractor_id = c.id
LEFT JOIN vagg ON vagg.contractor_id = c.id
LEFT JOIN hagg ON hagg.contractor_id = c.id;

CREATE UNIQUE INDEX IF NOT EXISTS contractor_risk_pk ON contractor_risk (contractor_id);
```
