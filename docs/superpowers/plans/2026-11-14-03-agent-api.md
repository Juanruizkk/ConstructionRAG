# Agent + API Implementation Plan (Plan 3 of 5)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans. Steps use checkbox (`- [ ]`) syntax.

**Goal:** Expose the data through a Fastify API and an LLM agent: typed tools (zod → services) that return every number from SQL, a RAG tool for narratives, and an agent loop (Groq behind an `LLMProvider`) with a strict citation contract.

**Architecture:** `apps/api/src/services` holds business logic over `@ssi/db` repositories. `apps/api/src/tools` are thin wrappers exposing zod schemas + handlers that call services and return `{ data, citations }`. `apps/api/src/agent` runs the tool-calling loop (max 6 iterations; tool + zod validation errors fed back to the LLM). `apps/api/src/server.ts` is the Fastify app. Numbers always come from tools/SQL; RAG only retrieves narrative text; `search_incidents` uses an exact filtered scan when a contractor is given and HNSW for global search, with a similarity threshold below which it returns "no relevant incidents".

**Tech Stack:** Fastify, zod, `zod-to-json-schema`, `groq-sdk`, `@ssi/db`, `@huggingface/transformers` (reused EmbeddingProvider), vitest. Prereq: Plans 1–2 complete, DB populated, `GROQ_API_KEY` set.

```ts
import Groq from 'groq-sdk';
import type { AssistantTurn, ChatMessage, LLMProvider, ToolSpec } from './llm-provider.js';

export class GroqProvider implements LLMProvider {
  readonly name = 'groq:llama-3.3-70b-versatile';
  private client = new Groq({ apiKey: process.env.GROQ_API_KEY });

  async complete(messages: ChatMessage[], tools: ToolSpec[]): Promise<AssistantTurn> {
    const res = await this.client.chat.completions.create({
      model: 'llama-3.3-70b-versatile',
      messages: messages.map((m) => ({
        role: m.role,
        content: m.content,
        ...(m.toolCallId ? { tool_call_id: m.toolCallId } : {}),
        ...(m.toolCalls ? { tool_calls: m.toolCalls.map((t) => ({ id: t.id, type: 'function', function: { name: t.name, arguments: t.arguments } })) } : {}),
      })) as never,
      tools: tools.map((t) => ({ type: 'function', function: { name: t.name, description: t.description, parameters: t.parameters } })),
      tool_choice: 'auto',
      temperature: 0,
    });
    const choice = res.choices[0]!.message;
    return {
      content: choice.content ?? '',
      toolCalls: (choice.tool_calls ?? []).map((c) => ({ id: c.id, name: c.function.name, arguments: c.function.arguments })),
    };
  }
}
```

```ts
import { sql } from 'drizzle-orm';
import type { Database } from '../client.js';

export interface RiskRow {
  contractor_id: number; display_name: string; risk_raw: number; risk_score: number;
  n_inspections: number; n_inspections_clean: number;
  n_serious: number; n_willful: number; n_repeat: number; n_other: number; n_unclassified: number;
  n_pending_contested: number; total_current_penalty: string;
  n_fatalities: number; n_fatalities_cited: number; n_fatalities_no_citation: number;
  n_hospitalizations: number; last_inspection_date: string | null; score_as_of: string;
}

export class RiskRepository {
  constructor(private readonly db: Database) {}

  async summary(contractorId: number): Promise<RiskRow | null> {
    const rows = await this.db.execute<RiskRow>(sql`
      SELECT r.*, c.display_name FROM contractor_risk r
      JOIN contractor c ON c.id = r.contractor_id WHERE r.contractor_id = ${contractorId}`);
    return (rows as unknown as RiskRow[])[0] ?? null;
  }

  async search(name: string, state?: string): Promise<{ contractor_id: number; display_name: string; confidence: number }[]> {
    const norm = name.toUpperCase().replace(/[^A-Z0-9 ]+/g, ' ').replace(/\s+/g, ' ').trim();
    const rows = await this.db.execute(sql`
      SELECT c.id AS contractor_id, c.display_name,
             similarity(c.norm_name, ${norm}) AS confidence
      FROM contractor c
      WHERE similarity(c.norm_name, ${norm}) > 0.3
        ${state ? sql`AND c.primary_state = ${state}` : sql``}
      ORDER BY confidence DESC LIMIT 10`);
    return rows as unknown as { contractor_id: number; display_name: string; confidence: number }[];
  }

  async compare(ids: number[], metric: string): Promise<{ contractor_id: number; display_name: string; value: number }[]> {
    const allowed: Record<string, string> = {
      risk_raw: 'r.risk_raw', fatalities: 'r.n_fatalities', serious: 'r.n_serious',
      repeat: 'r.n_repeat', willful: 'r.n_willful', total_penalty: 'r.total_current_penalty',
    };
    const col = allowed[metric] ?? 'r.risk_raw';
    const rows = await this.db.execute(sql`
      SELECT r.contractor_id, c.display_name, ${sql.raw(col)}::numeric AS value
      FROM contractor_risk r JOIN contractor c ON c.id = r.contractor_id
      WHERE r.contractor_id = ANY(${ids}) ORDER BY value DESC`);
    return rows as unknown as { contractor_id: number; display_name: string; value: number }[];
  }
}
```

```ts
import { createClient, RiskRepository } from '@ssi/db';
import { sql } from 'drizzle-orm';

export class ContractorService {
  private deps = createClient();
  private risk = new RiskRepository(this.deps.db);

  async search(name: string, state?: string) { return this.risk.search(name, state); }
  async summary(contractorId: number) { return this.risk.summary(contractorId); }
  async compare(ids: number[], metric: string) {
    if (metric === 'falls') {
      const rows = await this.deps.db.execute(sql`
        SELECT i.contractor_id, c.display_name,
               COUNT(*) FILTER (WHERE ai.fall_ht IS NOT NULL OR ai.fat_cause IS NOT NULL) AS value
        FROM accident_injury ai JOIN inspection i ON i.activity_nr = ai.rel_insp_nr
        JOIN contractor c ON c.id = i.contractor_id
        WHERE i.contractor_id = ANY(${ids}) GROUP BY 1,2 ORDER BY value DESC`);
      return rows as unknown as { contractor_id: number; display_name: string; value: number }[];
    }
    return this.risk.compare(ids, metric);
  }
  async listViolations(contractorId: number, since?: string, type?: string) {
    const rows = await this.deps.db.execute(sql`
      SELECT v.activity_nr, v.citation_id, v.standard, v.viol_type, v.current_penalty,
             v.issuance_date, (v.final_order_date IS NULL) AS pending
      FROM violation v JOIN inspection i ON i.activity_nr = v.activity_nr
      WHERE i.contractor_id = ${contractorId}
        ${since ? sql`AND v.issuance_date >= ${since}` : sql``}
        ${type ? sql`AND v.viol_type = ${type}` : sql``}
      ORDER BY v.issuance_date DESC LIMIT 100`);
    return rows as unknown as Record<string, unknown>[];
  }
}
```

```ts
import { createClient } from '@ssi/db';
import { sql } from 'drizzle-orm';
import { TransformersEmbeddingProvider } from '../embeddings/transformers-provider.js';

const THRESHOLD = 0.35; // cosine similarity floor (config-tunable)

export interface IncidentHit {
  summary_nr: number; event_date: string | null; text: string;
  similarity: number; contractor_ids: number[];
}

export class IncidentService {
  private deps = createClient();
  private embedder = new TransformersEmbeddingProvider();

  async search(query: string, contractorId?: number, k = 5): Promise<IncidentHit[]> {
    const [vec] = await this.embedder.embed([query]);
    const vecLit = `[${vec!.join(',')}]`;
    // With a contractor filter: exact filtered scan (small set, HNSW could drop matches).
    // Global: rely on HNSW index. Both compute cosine similarity = 1 - distance.
    const rows = await this.deps.db.execute(sql`
      SELECT summary_nr, event_date, text, contractor_ids,
             1 - (embedding <=> ${vecLit}::vector) AS similarity
      FROM accident_chunk
      WHERE embedding IS NOT NULL
        ${contractorId ? sql`AND contractor_ids @> ARRAY[${contractorId}]::int[]` : sql``}
      ORDER BY embedding <=> ${vecLit}::vector
      LIMIT ${k}`);
    const hits = rows as unknown as IncidentHit[];
    return hits.filter((h) => Number(h.similarity) >= THRESHOLD);
  }
}
```

```ts
import { IncidentService } from '../services/incident.service.js';
import { BidService } from '../services/bid.service.js';
import { oshaUrl, type Citation, type Tool } from './types.js';

const contractor = new ContractorService();
const incidents = new IncidentService();
const bids = new BidService();

export function buildTools(bidId: number): Tool[] {
  return [
    {
      name: 'get_bid',
      description: "List the subcontractors on the current bid (matched + not_found).",
      schema: z.object({}),
      async execute() { return { data: await bids.getBid(bidId), citations: [] }; },
    },
    {
      name: 'search_contractor',
      description: 'Find contractors by name. Returns candidates with a confidence score.',
      schema: z.object({ name: z.string(), state: z.string().length(2).optional() }),
      async execute(a) { return { data: await contractor.search(a.name, a.state), citations: [] }; },
    },
    {
      name: 'get_contractor_summary',
      description: 'Risk breakdown for one contractor (counts, penalties, fatalities, risk score).',
      schema: z.object({ contractor_id: z.number().int() }),
      async execute(a) {
        const s = await contractor.summary(a.contractor_id);
        const cits: Citation[] = [];
        return { data: s, citations: cits };
      },
    },
    {
      name: 'list_violations',
      description: 'List citation items for a contractor (optionally since a date / by type S|W|R|O|U).',
      schema: z.object({ contractor_id: z.number().int(), since: z.string().optional(), type: z.string().optional() }),

    {
      name: 'compare_contractors',
      description: 'Rank contractors by a metric: risk_raw|fatalities|serious|repeat|willful|total_penalty|falls.',
```

```ts
import type { LLMProvider, ChatMessage, ToolSpec } from './llm-provider.js';
import type { Citation, Tool } from '../tools/types.js';
import { zodToJsonSchema } from 'zod-to-json-schema';

const MAX_ITERS = 6;

const SYSTEM = (bidName: string) => `You are a construction safety assistant for a general contractor.
The current bid is "${bidName}". When the user says "my subs" call get_bid first.
RULES:
- Numbers (counts, penalties, fatalities, risk scores) MUST come from tool results. Never invent them.
- If the tools do not have the answer, say so plainly. Do not fabricate.
- Cite your sources: reference the activity_nr / incident returned by tools.
- If a contractor is not found, say "No OSHA record found — this does NOT mean the sub is safe."`;

export interface AgentInput {
  provider: LLMProvider; tools: Tool[]; bidName: string; userMessage: string;
  history?: ChatMessage[];
}
export interface AgentOutput { answer: string; citations: Citation[]; toolTrace: { name: string; args: string }[] }

export async function runAgent(input: AgentInput): Promise<AgentOutput> {
  const toolSpecs: ToolSpec[] = input.tools.map((t) => ({
    name: t.name, description: t.description,
    parameters: zodToJsonSchema(t.schema, { target: 'openApi3' }) as Record<string, unknown>,
  }));
  const byName = new Map(input.tools.map((t) => [t.name, t]));
  const messages: ChatMessage[] = [
    { role: 'system', content: SYSTEM(input.bidName) },
    ...(input.history ?? []),
    { role: 'user', content: input.userMessage },
  ];
  const citations: Citation[] = [];
  const toolTrace: { name: string; args: string }[] = [];

  for (let i = 0; i < MAX_ITERS; i++) {
    const turn = await input.provider.complete(messages, toolSpecs);
    if (turn.toolCalls.length === 0) {
      return { answer: turn.content, citations: dedup(citations), toolTrace };
    }
    messages.push({ role: 'assistant', content: turn.content, toolCalls: turn.toolCalls });
    for (const call of turn.toolCalls) {
      toolTrace.push({ name: call.name, args: call.arguments });
      const tool = byName.get(call.name);
      let resultContent: string;
      if (!tool) {
        resultContent = JSON.stringify({ error: `unknown tool ${call.name}` });
      } else {
        try {
          const parsed = tool.schema.parse(JSON.parse(call.arguments || '{}'));
          const result = await tool.execute(parsed);
          citations.push(...result.citations);
          resultContent = JSON.stringify(result.data);
        } catch (err) {
          resultContent = JSON.stringify({ error: err instanceof Error ? err.message : String(err) });
        }
      }
      messages.push({ role: 'tool', content: resultContent, toolCallId: call.id });
    }
  }
  // budget exhausted: ask for a final summary without tools
  const final = await input.provider.complete(
    [...messages, { role: 'user', content: 'Summarize your answer now using only the tool results above.' }], []);
  return { answer: final.content, citations: dedup(citations), toolTrace };
}

function dedup(cits: Citation[]): Citation[] {
  const seen = new Set<string>();
  return cits.filter((c) => { const k = JSON.stringify(c); if (seen.has(k)) return false; seen.add(k); return true; });
}
```

```ts
export function toBidRowDto(r: Record<string, unknown>): BidRowDto {
  return {
    contractorId: (r['contractor_id'] as number) ?? null,
    name: (r['display_name'] as string) ?? (r['entered_name'] as string),
    status: r['status'] as string,
    riskRaw: r['risk_raw'] != null ? Number(r['risk_raw']) : null,
    riskScore: r['risk_score'] != null ? Number(r['risk_score']) : null,
    fatalities: r['n_fatalities'] != null ? Number(r['n_fatalities']) : null,
    serious: r['n_serious'] != null ? Number(r['n_serious']) : null,
    inspections: r['n_inspections'] != null ? Number(r['n_inspections']) : null,
  };
}
```

```ts
import Fastify from 'fastify';
import { z } from 'zod';
import { BidService } from './services/bid.service.js';
import { ContractorService } from './services/contractor.service.js';
import { toBidRowDto } from './mappers/bid.mapper.js';
import { GroqProvider } from './agent/groq-provider.js';
import { runAgent } from './agent/loop.js';
import { buildTools } from './tools/index.js';

export function buildServer() {
  const app = Fastify({ logger: true });
  const bids = new BidService();
  const contractors = new ContractorService();
  const provider = new GroqProvider();

  app.get('/health', async () => ({ ok: true }));

  app.get('/bids/:id', async (req) => {
    const id = Number((req.params as { id: string }).id);
    const rows = await bids.getBid(id);
    return rows.map(toBidRowDto);
  });

  app.get('/contractors/search', async (req) => {
    const q = z.object({ name: z.string(), state: z.string().optional() }).parse(req.query);
    return contractors.search(q.name, q.state);
  });

  app.get('/contractors/:id/summary', async (req) => {
    const id = Number((req.params as { id: string }).id);
    return contractors.summary(id);
  });

  app.post('/chat', async (req) => {
    const body = z.object({ bidId: z.number(), bidName: z.string().default('current bid'), message: z.string() }).parse(req.body);
    const tools = buildTools(body.bidId);
    return runAgent({ provider, tools, bidName: body.bidName, userMessage: body.message });
  });

  return app;
}

if (import.meta.url === `file://${process.argv[1]}`) {
  const app = buildServer();
  app.listen({ port: Number(process.env.PORT ?? 3001), host: '0.0.0.0' })
    .catch((e) => { app.log.error(e); process.exit(1); });
}
```
